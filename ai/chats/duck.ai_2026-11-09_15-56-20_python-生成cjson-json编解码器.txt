import json
import os
import re
import sys
from dataclasses import dataclass, field

# ---------- Naming helpers ----------

C_KEYWORDS = {
    "auto", "break", "case", "char", "const", "continue", "default",
    "do", "double", "else", "enum", "extern", "float", "for", "goto",
    "if", "int", "long", "register", "return", "short", "signed",

def snake_case(value: str) -> str:
    value = re.sub(r"[^a-zA-Z0-9_]", "_", value)
    value = re.sub(r"([a-z0-9])([A-Z])", r"\1_\2", value)
    value = value.lower()

    if not value:
        value = "field"

    if value[0].isdigit():
        value = "_" + value

    if value in C_KEYWORDS:
        value += "_field"

    return value


def pascal_case(value: str) -> str:
    parts = re.split(r"[^a-zA-Z0-9]+", value)
    result = "".join(part[:1].upper() + part[1:] for part in parts if part)

    if not result:
        result = "Root"

    if result[0].isdigit():
        result = "T" + result

    return result


def c_string(value: str) -> str:
    return json.dumps(value)

    kind: str
    name: str | None = None
    fields: list["FieldNode"] = field(default_factory=list)

    type_node: TypeNode

        self.root_name = pascal_case(root_name)

        self.used_type_names: set[str] = set()

    def unique_type_name(self, name: str) -> str:
        base = pascal_case(name)
        result = base
        index = 2

        while result in self.used_type_names:
            result = f"{base}{index}"
            index += 1

        self.used_type_names.add(result)
        return result

    def infer_type(self, value: Any, suggested_name: str) -> TypeNode:
        if isinstance(value, bool):
            return TypeNode("bool")

        if isinstance(value, int) or isinstance(value, float):
            return TypeNode("number")

        if isinstance(value, str):
            return TypeNode("string")

        if isinstance(value, dict):
            type_name = self.unique_type_name(suggested_name)
            node = TypeNode("object", name=type_name)
            self.object_types.append(node)

            used_fields = set()

            for json_name, field_value in value.items():
                c_name = snake_case(json_name)

                # Avoid duplicate C field names
                original_name = c_name
                counter = 2
                while c_name in used_fields:
                    c_name = f"{original_name}_{counter}"
                    counter += 1

                used_fields.add(c_name)

                child_type = self.infer_type(
                    field_value,
                    suggested_name=f"{type_name}_{pascal_case(json_name)}"
                )

                node.fields.append(
                    FieldNode(
                        json_name=json_name,
                        c_name=c_name,
                        type_node=child_type,
                    )
                )

            return node

        if isinstance(value, list):
            if not value:
                raise ValueError(
                    f"Empty array found for '{suggested_name}'. "
                    "Use a non-empty representative array."
                )

            first_type = self.infer_type(

            )

                    raise ValueError(

                    )

            return TypeNode("array", element=first_type)

        if value is None:
            raise ValueError(
                f"JSON null found at '{suggested_name}'. "
                "This generator does not support null values."
            )

        raise ValueError(
            f"Unsupported JSON value at '{suggested_name}': {type(value)}"
        )

    def same_type(self, left: TypeNode, right: TypeNode) -> bool:
        if left.kind != right.kind:
            return False

        if left.kind == "object":
            if len(left.fields) != len(right.fields):
                return False

            for lf, rf in zip(left.fields, right.fields):
                if lf.json_name != rf.json_name:
                    return False
                if not self.same_type(lf.type_node, rf.type_node):
                    return False

        return True

    def c_base_type(self, node: TypeNode) -> str:
        if node.kind == "string":
            return "char *"

        if node.kind == "number":
            return "double"

        if node.kind == "bool":
            return "bool"

        if node.kind == "object":

        output.append(f"#ifndef {guard}")

        output.append(f"void {lower_root}_free({root} *value);")
        output.append("")
        output.append("#endif")

        return "\n".join(output) + "\n"

        output = []

        output.append(f'#include "{header_name}"')

        output.append("#include <string.h>")

        output.append("    }")
        output.append("    return copy;")
        output.append("}")
        output.append("")

        # Internal function prototypes
        for node in self.object_types:
            output.append(
                f"static cJSON *encode_{node.name}(const {node.name} *value);"
            )
            output.append(
                f"static int decode_{node.name}("
                f"const cJSON *json, {node.name} *value);"
            )
            output.append(
                f"static void free_{node.name}({node.name} *value);"
            )
            output.append("")

        # Internal object functions

            output.extend(self.generate_encode_function(node))
            output.append("")
            output.extend(self.generate_decode_function(node))
            output.append("")
            output.extend(self.generate_free_function(node))
            output.append("")

        # Public root wrappers
        root = self.root_name
        lower_root = snake_case(root)

        output.append(
            f"cJSON *{lower_root}_encode(const {root} *value)"
        )
        output.append("{")
        output.append("    return encode_" + root + "(value);")
        output.append("}")
        output.append("")

        output.append(

        output.append("{")
        output.append("    return decode_" + root + "(json, value);")
        output.append("}")
        output.append("")

        output.append(f"void {lower_root}_free({root} *value)")
        output.append("{")
        output.append("    free_" + root + "(value);")
        output.append("}")

        return "\n".join(output) + "\n"
